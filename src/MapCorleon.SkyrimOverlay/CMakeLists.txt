cmake_minimum_required(VERSION 3.25)
project(MapCorleonOverlay VERSION 0.1.0 LANGUAGES CXX)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the Skyrim overlay with the MSVC x64 toolchain.")
endif()

include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
set(FMT_INSTALL ON CACHE BOOL "" FORCE)

FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 10.2.1 GIT_SHALLOW TRUE OVERRIDE_FIND_PACKAGE)
FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG v1.13.0 GIT_SHALLOW TRUE OVERRIDE_FIND_PACKAGE)
FetchContent_Declare(rapidcsv GIT_REPOSITORY https://github.com/d99kris/rapidcsv.git GIT_TAG v8.84 GIT_SHALLOW TRUE SOURCE_SUBDIR include)
FetchContent_Declare(commonlib GIT_REPOSITORY https://github.com/CharmedBaryon/CommonLibSSE-NG.git GIT_TAG v3.7.0 GIT_SHALLOW TRUE)
FetchContent_Declare(menu_api GIT_REPOSITORY https://github.com/QTR-Modding/SKSE-Menu-Framework-3-API.git GIT_TAG 1dcb70179076aae4ab626f43c5baab2735ca5877)
FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)

FetchContent_MakeAvailable(fmt spdlog rapidcsv menu_api nlohmann_json)
set(RAPIDCSV_INCLUDE_DIRS "${rapidcsv_SOURCE_DIR}/src" CACHE PATH "")
FetchContent_MakeAvailable(commonlib)
add_library(MapCorleonOverlay SHARED Overlay.cpp PipeClient.cpp)
target_include_directories(MapCorleonOverlay SYSTEM PRIVATE "${menu_api_SOURCE_DIR}" "${RAPIDCSV_INCLUDE_DIRS}")
target_link_libraries(MapCorleonOverlay PRIVATE CommonLibSSE::CommonLibSSE nlohmann_json::nlohmann_json)
target_compile_definitions(MapCorleonOverlay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_precompile_headers(MapCorleonOverlay PRIVATE <SKSE/Impl/PCH.h>)
install(TARGETS MapCorleonOverlay RUNTIME DESTINATION SKSE/Plugins COMPONENT Overlay)
install(FILES MapCorleonOverlay.ini DESTINATION SKSE/Plugins COMPONENT Overlay)
